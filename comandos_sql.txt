BEGIN;

-- Criar tabela de Usuários (CustomUser)
CREATE TABLE IF NOT EXISTS api_customuser (
    id BIGSERIAL PRIMARY KEY,
    password VARCHAR(128) NOT NULL,
    last_login WITH TIME ZONE,
    is_superuser BOOLEAN NOT NULL DEFAULT FALSE,
    first_name VARCHAR(150) NOT NULL,
    last_name VARCHAR(150) NOT NULL,
    email VARCHAR(254) NOT NULL,
    is_staff BOOLEAN NOT NULL DEFAULT FALSE,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    date_joined WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    employee_id VARCHAR(50) UNIQUE NOT NULL,
    role VARCHAR(10) NOT NULL DEFAULT 'employee'
);

-- Criar tabela de Registros de Ponto (TimeRecord)
CREATE TABLE IF NOT EXISTS api_timerecord (
    id BIGSERIAL PRIMARY KEY,
    date DATE NOT NULL,
    clock_in WITH TIME ZONE,
    clock_out WITH TIME ZONE,
    observation TEXT,
    created_at WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    user_id BIGINT NOT NULL,
    CONSTRAINT fk_user FOREIGN KEY (user_id) REFERENCES api_customuser(id) ON DELETE CASCADE,
    UNIQUE (user_id, date)
);
